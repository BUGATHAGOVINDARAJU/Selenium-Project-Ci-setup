HEADLESS=False             # Set True for background execution
IMPLICIT_WAIT=10           # Seconds
EXPLICIT_WAIT=20           # Seconds
PAGE_LOAD_TIMEOUT=30       # Seconds
OUTPUT_FORMAT=json         # Options: json, csv, excel
OUTPUT_PATH=./output
MAX_PAGES=5               # Maximum pages to scrape

TROUBLESHOOTING
================================================================================

Issue: Module not found
  Fix: venv\Scripts\activate && pip install -r requirements.txt

Issue: Browser doesn't open
  Fix: Ensure Chrome/Firefox/Edge is installed
       Try different browser in .env file

Issue: No jobs found
  Fix: Set HEADLESS=False in .env to see browser actions
       Check if Indeed is accessible in your region

Issue: Element not found
  Fix: Increase timeout values in .env file
       Check logs/ folder for detailed error information
